#include "engine.hpp"
#include <cmath>
#include <numbers>
#include <algorithm>

constexpr double EARTH_RADIUS = 6378137.0;
constexpr double DEG_TO_RAD = std::numbers::pi / 180.0;
constexpr double RAD_TO_DEG = 180.0 / std::numbers::pi;
constexpr double FT_TO_M = 0.3048;
constexpr double M_TO_FT = 1.0 / 0.3048;
constexpr double KTS_TO_MS = 0.514444;

// Altitude-based wind interpolation (Altitude in Feet, Speed in m/s)
static std::pair<double, double> interpolate_alt(const std::map<double, std::pair<double, double>>& wind, double alt_ft, double lz_alt_ft) {
    if (alt_ft <= lz_alt_ft || wind.empty()) return {0.0, 0.0};

    auto upper_it = wind.upper_bound(alt_ft);
    if (upper_it == wind.end()) return std::prev(wind.end())->second;

    double alt1, S1, D1;
    if (upper_it == wind.begin()) {
        alt1 = lz_alt_ft; S1 = 0.0; D1 = 0.0;
    } else {
        auto lower_it = std::prev(upper_it);
        alt1 = lower_it->first; S1 = lower_it->second.first; D1 = lower_it->second.second;
    }

    double alt2 = upper_it->first; double S2 = upper_it->second.first; double D2 = upper_it->second.second;

    double U1 = -S1 * std::sin(D1 * DEG_TO_RAD); double V1 = -S1 * std::cos(D1 * DEG_TO_RAD);
    double U2 = -S2 * std::sin(D2 * DEG_TO_RAD); double V2 = -S2 * std::cos(D2 * DEG_TO_RAD);

    double t = (alt_ft - alt1) / (alt2 - alt1);
    double U = U1 + t * (U2 - U1); double V = V1 + t * (V2 - V1);

    double S = std::sqrt(U * U + V * V);
    double D = std::atan2(-U, -V) * RAD_TO_DEG;
    D = std::fmod(D, 360.0);
    if (D < 0.0) D += 360.0;

    return {S, D};
}

// Temporary struct for the relative Cartesian shifting
struct RelPoint { double x, y, alt; };

void run_physics(
    std::array<std::vector<geo_point>, 360>& path_list,
    const std::unordered_map<std::string, double>& params,
    const std::map<double, std::pair<double, double>>& wind
) {
    double LZ_LAT = params.at("LZ LAT");
    double LZ_LON = params.at("LZ LON");
    double lz_alt_m = params.at("LZ ELEV"); // Assuming UI provides meters

    double jump_alt_m = params.at("JUMP ALT") * FT_TO_M;
    double open_alt_m = params.at("OPEN ALT") * FT_TO_M;

    double airv_ms = params.at("AC VEL KTS") * KTS_TO_MS;
    double airh_deg = params.at("AC HDG DEG");

    double canopy_v_ms = params.at("0-WIND CANOPY VEL") * KTS_TO_MS;
    double gr = params.at("GR");
    if (gr <= 0) gr = 1.0;
    double descv_ms = canopy_v_ms / gr;

    double vz_ff_ms = 45.0; // Freefall vertical velocity (m/s)
    double dz = 5.0;        // 5-meter integration steps

    // Master Cartesian Coordinate definition: +X is East, +Y is North
    double ac_vx = airv_ms * std::sin(airh_deg * DEG_TO_RAD);
    double ac_vy = airv_ms * std::cos(airh_deg * DEG_TO_RAD);

    // --- PHASE 1: Freefall Trajectory (Jump -> Open) ---
    // Start directly over the LZ (0,0)
    std::vector<RelPoint> freefall_path;
    double cx = 0.0, cy = 0.0;
    freefall_path.push_back({cx, cy, jump_alt_m});

    for (double alt = jump_alt_m; alt > open_alt_m; alt -= dz) {
        auto [w_ms, w_deg] = interpolate_alt(wind, alt * M_TO_FT, lz_alt_m * M_TO_FT);

        // Meteorological wind points 'to' opposite direction
        double wind_vx = -w_ms * std::sin(w_deg * DEG_TO_RAD);
        double wind_vy = -w_ms * std::cos(w_deg * DEG_TO_RAD);

        cx += (wind_vx + ac_vx) * (dz / vz_ff_ms);
        cy += (wind_vy + ac_vy) * (dz / vz_ff_ms);

        ac_vx *= 0.95; ac_vy *= 0.95; // Decay aircraft momentum

        double next_alt = std::max(alt - dz, open_alt_m);
        freefall_path.push_back({cx, cy, next_alt});
    }

    double ff_end_x = cx;
    double ff_end_y = cy;

    // --- PHASE 2: Canopy Trajectory (Open -> LZ) & Parallel Shift ---
    for (int i = 0; i < 360; ++i) {
        path_list[i].clear();

        std::vector<RelPoint> canopy_path;
        double px = ff_end_x;
        double py = ff_end_y;
        double h_rad = i * DEG_TO_RAD;

        // Calculate the forward canopy path for this specific heading
        for (double alt = open_alt_m; alt > lz_alt_m; alt -= dz) {
            auto [w_ms, w_deg] = interpolate_alt(wind, alt * M_TO_FT, lz_alt_m * M_TO_FT);

            double wind_vx = -w_ms * std::sin(w_deg * DEG_TO_RAD);
            double wind_vy = -w_ms * std::cos(w_deg * DEG_TO_RAD);

            double glide_vx = canopy_v_ms * std::sin(h_rad);
            double glide_vy = canopy_v_ms * std::cos(h_rad);

            px += (wind_vx + glide_vx) * (dz / descv_ms);
            py += (wind_vy + glide_vy) * (dz / descv_ms);

            double next_alt = std::max(alt - dz, lz_alt_m);
            canopy_path.push_back({px, py, next_alt});
        }

        // --- PHASE 3: The Parallel Shift ---
        // (px, py) is where we landed. We want the landing to be exactly at the LZ (0,0).
        // Therefore, the whole trajectory needs to be shifted by (-px, -py).
        double shift_x = -px;
        double shift_y = -py;

        // Map the shifted Cartesian coords to true Lat/Lon
        // A. Shift and map Freefall leg
        for (const auto& pt : freefall_path) {
            double sx = pt.x + shift_x;
            double sy = pt.y + shift_y;

            double lat = LZ_LAT + (sy / EARTH_RADIUS) * RAD_TO_DEG;
            double lon = LZ_LON + (sx / (EARTH_RADIUS * std::cos(lat * DEG_TO_RAD))) * RAD_TO_DEG;
            path_list[i].push_back(geo_point{lat, lon, pt.alt});
        }

        // B. Shift and map Canopy leg
        for (const auto& pt : canopy_path) {
            double sx = pt.x + shift_x;
            double sy = pt.y + shift_y;

            double lat = LZ_LAT + (sy / EARTH_RADIUS) * RAD_TO_DEG;
            double lon = LZ_LON + (sx / (EARTH_RADIUS * std::cos(lat * DEG_TO_RAD))) * RAD_TO_DEG;
            path_list[i].push_back(geo_point{lat, lon, pt.alt});
        }
    }
}

#include "points.hpp"
#include <cmath>
#include <numbers>

inline constexpr double R0 = 6356766; // meters; 45 32 33 latitude
inline constexpr double PI = std::numbers::pi;
inline constexpr double DEG_TO_RAD = std::numbers::pi / 180.0;
inline constexpr double RAD_TO_DEG = 180.0 / std::numbers::pi;

// geo_point -> cartesian_point definition
geo_point::operator cartesian_point() const {
    cartesian_point p;
    const double rad_lat = lat * DEG_TO_RAD;
    const double rad_lon = lon * DEG_TO_RAD;
    const double r = R0 + alt;

    p.x = r * std::cos(rad_lat) * std::cos(rad_lon);
    p.y = r * std::cos(rad_lat) * std::sin(rad_lon);
    p.z = r * std::sin(rad_lat);
    return p;
}

// cartesian_point -> geo_point definition
cartesian_point::operator geo_point() const {
    geo_point gp;
    double r = std::sqrt(x * x + y * y);
    gp.lon = std::atan2(y, x) * RAD_TO_DEG;
    gp.lat = std::atan2(z, r) * RAD_TO_DEG;
    gp.alt = r - R0;
    return gp;
}

    cartesian_point operator-(const cartesian_point &p) const {

void MissionController::runMission(const QVariantMap& missionData) {
    params.clear();
    wind.clear();

    params["LZ LAT"] = missionData["lzLat"].toDouble();
    params["LZ LON"] = missionData["lzLon"].toDouble();
    params["LZ ELEV"] = missionData["lzElev"].toDouble();
    params["JUMP ALT"] = missionData["jumpAlt"].toDouble();
    params["OPEN ALT"] = missionData["openAlt"].toDouble(); // CRITICAL: Ensure this is uncommented
    params["WT KG"] = missionData["weight"].toDouble();
    params["AC VEL KTS"] = missionData["acVel"].toDouble();
    params["AC HDG DEG"] = missionData["acHdg"].toDouble();
    params["0-WIND CANOPY VEL"] = missionData["canopyVel"].toDouble();
    params["GR"] = missionData["glideRatio"].toDouble();

    QVariantList windList = missionData["windProfile"].toList();
    for (const QVariant& vRow : windList) {
        QVariantMap row = vRow.toMap();
        double alt = row["altitude"].toDouble();
        double spd = row["speed"].toDouble();
        double hdg = row["heading"].toDouble();
        wind[alt] = std::make_pair(spd, hdg);
    }

    run_physics(path_lists, params, wind);

    // Extract the VERY TOP Jump Point (Aircraft Release point) to draw the locus
    QVariantList perimeter;
    for (int i = 0; i < 360; ++i) {
        if (!path_lists[i].empty()) {
            QVariantMap pt;
            pt["lat"] = path_lists[i][0].lat; // <--- Index 0 is the start of the parallel-shifted path
            pt["lon"] = path_lists[i][0].lon;
            perimeter.append(pt);
        }
    }

    emit envelopeCalculated(perimeter);
}

void MissionController::exportKml(const QString& filePath) {
    qDebug() << "KML Export triggered for:" << filePath;
    // TODO: Iterate through path_lists and generate KML XML structure
    QString localPath = QUrl(filePath).toLocalFile();
    if (localPath.isEmpty()) {
        localPath = filePath;
    }

    // 1. Load the external XML templates from the Qt Resource System
    QFile docFile(":/qt/qml/handle/kml_document.xml");
    QFile placemarkFile(":/qt/qml/handle/kml_placemark.xml");

    if (!docFile.open(QIODevice::ReadOnly | QIODevice::Text) ||
        !placemarkFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
        qWarning() << "Failed to load external KML XML templates.";
        return;
        }

    QString docTemplate = QTextStream(&docFile).readAll();
    QString placemarkTemplate = QTextStream(&placemarkFile).readAll();

    docFile.close();
    placemarkFile.close();

    // 2. Generate the internal Placemarks
    QString allPlacemarks;
    for (int i = 0; i < 360; ++i) {
        if (path_lists[i].empty()) continue;

        QString coordinates;
        for (const auto& pt : path_lists[i]) {
            // Build the coordinate block with strict spacing for clean XML output
            coordinates += QString("          %1,%2,%3\n")
                               .arg(pt.lon, 0, 'f', 6)
                               .arg(pt.lat, 0, 'f', 6)
                               .arg(pt.alt, 0, 'f', 2);
        }

        // Inject the heading (i) and coordinate list into the Placemark template
        allPlacemarks += placemarkTemplate.arg(i).arg(coordinates);
    }

    // 3. Inject all Placemarks into the main Document template
    QString finalKml = docTemplate.arg(allPlacemarks);

    // 4. Write the assembled KML to the user's hard drive
    QFile outFile(localPath);
    if (outFile.open(QIODevice::WriteOnly | QIODevice::Text)) {
        QTextStream out(&outFile);
        out << finalKml;
        outFile.close();
        qDebug() << "Successfully exported KML to:" << localPath;
    } else {
        qWarning() << "Could not open file for KML export:" << localPath;
    }
}

void MissionController::fetchWindProfile(double lat, double lon, const QString& date, const QString& time) {
    emit progressUpdated("Fetching Atmospheric Data...", 0.3);

    QtConcurrent::run([this, lat, lon, date, time]() {
        try {
            // 1. Generate current UTC date and exact hour for the API query
            // QDateTime now = QDateTime::currentDateTimeUtc();
            // QString date = now.toString("yyyy-MM-dd");
            // QString time = now.toString("HH:00");

            // 2. Execute your fetchwind.cpp HTTP request
            WindMap rawWind = fetchWind(lat, lon, date.toStdString(), time.toStdString());

            // 3. Clear the global powertrain hashmap to receive the new data
            wind.clear();
            QVariantList tableData;

            // 4. Convert Pressure (hPa) to Standard Altitude (Feet)
            for (const auto& [pressure, windData] : rawWind) {
                double speed = windData.first;
                double direction = windData.second;

                // Standard Barometric Formula: h = 145366.45 * (1 - (P / 1013.25)^0.190284)
                double altFeet = 145366.45 * (1.0 - std::pow(pressure / 1013.25, 0.190284));
                double roundedAlt = std::round(altFeet / 100.0) * 100.0; // Round to nearest 100ft

                // Populate the C++ Physics Hashmap
                wind[roundedAlt] = windData;

                // Populate the QML UI Row
                QVariantMap row;
                row["altitude"] = QString::number(roundedAlt, 'f', 0);
                row["speed"] = QString::number(speed, 'f', 1);
                row["heading"] = QString::number(direction, 'f', 0);
                tableData.append(row);
            }

            // 5. Sort the QML table data descending (Highest altitude at the top of the table)
            std::sort(tableData.begin(), tableData.end(), [](const QVariant& a, const QVariant& b) {
                return a.toMap()["altitude"].toDouble() > b.toMap()["altitude"].toDouble();
            });

            // 6. Push the updated data back to the QML Main Thread
            QMetaObject::invokeMethod(this, [this, tableData]() {
                emit windProfileFetched(tableData);
                emit progressUpdated("Wind Profile Downloaded", 1.0);
            }, Qt::QueuedConnection);

        } catch (const std::exception& e) {
            QMetaObject::invokeMethod(this, [this, e]() {
                emit errorReported(QString("Wind API Error: ") + e.what());
                emit progressUpdated("Failed", 1.0);
            }, Qt::QueuedConnection);
        }
    });
}

void MissionController::saveProject(const QString& filePath, const QVariantMap& projectData) {
    QString localPath = QUrl(filePath).toLocalFile();
    if (localPath.isEmpty()) localPath = filePath;

    QFile file(localPath);
    if (file.open(QIODevice::WriteOnly | QIODevice::Text)) {
        // Automatically converts the QML dictionary to a formatted JSON file
        QJsonDocument doc(QJsonObject::fromVariantMap(projectData));
        file.write(doc.toJson());
        file.close();
        qDebug() << "Project saved to:" << localPath;
    } else {
        emit errorReported("Failed to save project file.");
    }
}

QVariantMap MissionController::loadProject(const QString& filePath) {
    QString localPath = QUrl(filePath).toLocalFile();
    if (localPath.isEmpty()) localPath = filePath;

    QFile file(localPath);
    if (file.open(QIODevice::ReadOnly | QIODevice::Text)) {
        QJsonDocument doc = QJsonDocument::fromJson(file.readAll());
        file.close();
        return doc.object().toVariantMap(); // Returns JSON as a QML dictionary
    }
    emit errorReported("Failed to open project file.");
    return QVariantMap();
}

    settings.setValue("cachePath", localPath);

    // Ensure the new directory physically exists

}

size_t writeCallback(
    void* contents,
    size_t size,
    size_t nmemb,
    void* userdata
    ) {
    const size_t bytes = size * nmemb;
    auto* response = static_cast<std::string*>(userdata);
    response->append(
        static_cast<char*>(contents),
        bytes
        );
    return bytes;
}

std::string buildHourlyVariables() {
    std::string hourly;

    for (int pressure : pressure_levels) {
        if (!hourly.empty()) {
            hourly += ",";
        }

        hourly +=
            "wind_speed_" +
            std::to_string(pressure) +
            "hPa";

        hourly +=
            ",wind_direction_" +
            std::to_string(pressure) +
            "hPa";
    }

    return hourly;
}

std::string buildURL(
            double latitude,
            double longitude
        ) {

std::string httpGet(const std::string& url) {
            // Initialize libcurl once for this process.
            static const CURLcode initResult =
                curl_global_init(CURL_GLOBAL_DEFAULT);

            if (initResult != CURLE_OK) {
                throw std::runtime_error(
                    "Failed to initialize libcurl"
                );
            }

            CURL* curl = curl_easy_init();

            if (!curl) {
                throw std::runtime_error(
                    "Failed to create CURL handle"
                );
            }

            std::string response;

            curl_easy_setopt(
                curl,
                CURLOPT_URL,
                url.c_str()
            );

            curl_easy_setopt(
                curl,
                CURLOPT_WRITEFUNCTION,
                writeCallback
            );

            curl_easy_setopt(
                curl,
                CURLOPT_WRITEDATA,
                &response
            );

            curl_easy_setopt(
                curl,
                CURLOPT_CONNECTTIMEOUT,
                10L
            );

            curl_easy_setopt(
                curl,
                CURLOPT_TIMEOUT,
                30L
            );

            curl_easy_setopt(
                curl,
                CURLOPT_FOLLOWLOCATION,
                1L
            );

            const CURLcode result =
                curl_easy_perform(curl);

            long statusCode = 0;

            curl_easy_getinfo(
                curl,
                CURLINFO_RESPONSE_CODE,
                &statusCode
            );

            curl_easy_cleanup(curl);

            if (result != CURLE_OK) {
                throw std::runtime_error(
                    std::string("HTTP request failed: ") +
                    curl_easy_strerror(result)
                );
            }

            if (statusCode != 200) {
                throw std::runtime_error(
                    "Open-Meteo HTTP status " +
                    std::to_string(statusCode) +
                    ": " + response
                );
            }

            return response;
        }

WindMap fetchWind(
        double latitude,
        double longitude,
        const std::string& date,
        const std::string& time
    ) {
        if (!std::isfinite(latitude) ||
            latitude < -90.0 ||
            latitude > 90.0) {
            throw std::invalid_argument(
                "Latitude must be between -90 and 90"
            );
        }

        if (!std::isfinite(longitude) ||
            longitude < -180.0 ||
            longitude > 180.0) {
            throw std::invalid_argument(
                "Longitude must be between -180 and 180"
            );
        }

        if (date.size() != 10 ||
            date[4] != '-' ||
            date[7] != '-' ||
            time.size() != 5 ||
            time[2] != ':') {
            throw std::invalid_argument(
                "Expected date YYYY-MM-DD and time HH:MM"
            );
        }

        const std::string response = httpGet(
            buildURL(latitude, longitude)
        );

        const json data = json::parse(response);

        if (data.value("error", false)) {
            throw std::runtime_error(
                data.value(
                    "reason",
                    "Open-Meteo API error"
                )
            );
        }

        const auto& hourly = data.at("hourly");
        const auto& times = hourly.at("time");

        const std::string requestedTimestamp =
            date + "T" + time;

        size_t timeIndex = times.size();

        for (size_t i = 0; i < times.size(); ++i) {
            if (times[i].get<std::string>() ==
                requestedTimestamp) {
                timeIndex = i;
                break;
            }
        }

        if (timeIndex == times.size()) {
            throw std::runtime_error(
                "Timestamp not available in API response: " +
                requestedTimestamp
            );
        }

        WindMap wind;

        for (int pressure : pressure_levels) {
            const std::string speedKey =
                "wind_speed_" +
                std::to_string(pressure) +
                "hPa";

            const std::string directionKey =
                "wind_direction_" +
                std::to_string(pressure) +
                "hPa";

            const auto& speeds = hourly.at(speedKey);
            const auto& directions = hourly.at(directionKey);

            if (speeds[timeIndex].is_null() ||
                directions[timeIndex].is_null()) {
                continue;
            }

            const double speed =
                speeds[timeIndex].get<double>();

            const double direction =
                directions[timeIndex].get<double>();

            if (!std::isfinite(speed) ||
                !std::isfinite(direction)) {
                continue;
            }

            wind.emplace(
                static_cast<double>(pressure),
                std::make_pair(speed, direction)
            );
        }

        return wind;
    }

void MapCanvasItem::updateMission(double latitude, double longitude, double widthKm, double heightKm) {
    // 1. Clamp Latitude to Web Mercator limits (prevents Infinity math crashes at the poles)
    latitude = std::clamp(latitude, -85.0511, 85.0511);

    // 2. Wrap Longitude cleanly so crossing the International Date Line works perfectly
    longitude = std::fmod(longitude, 360.0);
    if (longitude >= 180.0) longitude -= 360.0;
    else if (longitude < -180.0) longitude += 360.0;

    // 3. Clamp camera view limits to prevent impossible map fetch requests
    widthKm = std::clamp(widthKm, 0.05, 5000.0);
    heightKm = std::clamp(heightKm, 0.05, 5000.0);

    m_centerLat = latitude;
    m_centerLon = longitude;
    m_viewWidthMeters = widthKm * 1000.0;
    m_viewHeightMeters = heightKm * 1000.0;

    if (!m_dataset) return;

    // [THE FIX]: Safely clamp the UTM zone strictly between 1 and 60
    int utmZone = static_cast<int>(std::floor((longitude + 180.0) / 6.0)) + 1;
    utmZone = std::clamp(utmZone, 1, 60);
    bool isNorth = (latitude >= 0.0);

    OGRSpatialReference latLonSrs, utmSrs;
    latLonSrs.SetWellKnownGeogCS("WGS84");
    latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
    utmSrs.SetWellKnownGeogCS("WGS84");
    utmSrs.SetUTM(utmZone, isNorth);

    char *dstWkt = nullptr;
    utmSrs.exportToWkt(&dstWkt);
    m_utmWkt = QString(dstWkt);
    CPLFree(dstWkt);

    OGRCoordinateTransformation *coordTrans = OGRCreateCoordinateTransformation(&latLonSrs, &utmSrs);
    double utmX = longitude;
    double utmY = latitude;
    if (coordTrans) {
        coordTrans->Transform(1, &utmX, &utmY);
        m_utmCenterX = utmX;
        m_utmCenterY = utmY;
        OCTDestroyCoordinateTransformation(coordTrans);
    }

    refreshMapBuffer(); // Generate the image in memory
    emit viewChanged();
}

void MapCanvasItem::refreshMapBuffer() {
    if (!m_dataset || m_viewWidthMeters <= 0 || m_viewHeightMeters <= 0 || m_utmWkt.isEmpty()) return;

    int screenWidth = boundingRect().width();
    int screenHeight = boundingRect().height();
    if (screenWidth <= 0 || screenHeight <= 0) return;

    // Force 1:1 Aspect Ratio
    double aspectRatio = static_cast<double>(screenWidth) / screenHeight;
    m_viewHeightMeters = m_viewWidthMeters / aspectRatio;

    // 1. Thread Tracking (Cancels older threads if user pans quickly)
    int fetchId = ++m_fetchCounter;
    emit progressUpdated("Loading Map Data...", 0.0);

    // 2. Snapshot current state by value
    double viewW = m_viewWidthMeters;
    double viewH = m_viewHeightMeters;
    double utmCX = m_utmCenterX;
    double utmCY = m_utmCenterY;
    QString utmWkt = m_utmWkt;

    // Load XML config dynamically for the background thread
    QFile xmlFile(":/qt/qml/ui/mapcanvas/osm_config.xml");
    if (!xmlFile.open(QIODevice::ReadOnly | QIODevice::Text)) return;
    // QString threadXmlConfig = QTextStream(&xmlFile).readAll().arg(QDir::currentPath() + "/map_cache");
    QString rawXml = QTextStream(&xmlFile).readAll();

    QSettings settings("Skytec", "MissionPlanner");
    QString defaultPath = QStandardPaths::writableLocation(QStandardPaths::AppDataLocation) + "/map_cache";
    QString cachePath = settings.value("cachePath", defaultPath).toString();
    QDir().mkpath(cachePath); // Guarantee folder exists before GDAL runs
    QString threadXmlConfig = rawXml.arg(cachePath);


    // 3. Launch Async Network Fetch
    QtConcurrent::run([this, fetchId, viewW, viewH, utmCX, utmCY, utmWkt, threadXmlConfig, screenWidth, screenHeight]() {
        GDALDriver *memDriver = GetGDALDriverManager()->GetDriverByName("MEM");
        QByteArray xmlBytes = threadXmlConfig.toUtf8();
        GDALDataset* threadDS = (GDALDataset*) GDALOpenEx(xmlBytes.constData(), GDAL_OF_RASTER, nullptr, nullptr, nullptr);

        if (!threadDS) return;

        // Coordinate Math
        OGRSpatialReference latLonSrs, utmSrs;
        latLonSrs.SetWellKnownGeogCS("WGS84");
        latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
        utmSrs.importFromWkt(utmWkt.toStdString().c_str());
        utmSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);

        OGRCoordinateTransformation *invTrans = OGRCreateCoordinateTransformation(&utmSrs, &latLonSrs);
        double currentLon = utmCX;
        double currentLat = utmCY;
        if (invTrans) {
            invTrans->Transform(1, &currentLon, &currentLat);
            OCTDestroyCoordinateTransformation(invTrans);
        }

        double webMercCenterX = currentLon * 20037508.34 / 180.0;
        double latRad = currentLat * M_PI / 180.0;
        double webMercCenterY = log(tan(M_PI / 4.0 + latRad / 2.0)) * 20037508.34 / M_PI;

        // double pad = 1.2;
        // double wmWidth = viewW * pad;
        // double wmHeight = viewH * pad;
        // double wmMinX = webMercCenterX - (wmWidth / 2.0);
        // double wmMaxY = webMercCenterY + (wmHeight / 2.0);
        // [THE FIX]: Account for Web Mercator latitude stretch
        double scaleFactor = 1.0 / std::cos(latRad);
        double pad = 1.2;

        double wmWidth = viewW * scaleFactor * pad;
        double wmHeight = viewH * scaleFactor * pad;

        double wmMinX = webMercCenterX - (wmWidth / 2.0);
        double wmMaxY = webMercCenterY + (wmHeight / 2.0);

        int wmPixWidth = screenWidth * pad;
        int wmPixHeight = screenHeight * pad;

        double wmsGeoTransform[6];
        threadDS->GetGeoTransform(wmsGeoTransform);
        int xOff = static_cast<int>((wmMinX - wmsGeoTransform[0]) / wmsGeoTransform[1]);
        int yOff = static_cast<int>((wmMaxY - wmsGeoTransform[3]) / wmsGeoTransform[5]);
        int dsXSize = static_cast<int>(wmWidth / wmsGeoTransform[1]);
        int dsYSize = static_cast<int>(std::abs(wmHeight / wmsGeoTransform[5]));

        std::vector<GByte> rawBuffer(wmPixWidth * wmPixHeight * 3, 255);
        int bands[3] = {1, 2, 3};

        // 4. CHUNKED NETWORK FETCH (Provides real-time UI Progress)
        int strips = 4;
        bool fetchFailed = false;
        for (int i = 0; i < strips; ++i) {
            if (fetchId != m_fetchCounter) break; // Early abort if user panned again

            int yOffStrip = yOff + (i * dsYSize / strips);
            int ySizeStrip = dsYSize / strips;
            int destYOff = i * wmPixHeight / strips;
            int destYSize = wmPixHeight / strips;

            CPLErr err = threadDS->RasterIO(GF_Read, xOff, yOffStrip, dsXSize, ySizeStrip,
                                            rawBuffer.data() + (destYOff * wmPixWidth * 3),
                                            wmPixWidth, destYSize,
                                            GDT_Byte, 3, bands, 3, wmPixWidth * 3, 1);
            if (err != CE_None) {
                fetchFailed = true;
                break;
            }
            emit progressUpdated("Fetching Live Map...", static_cast<double>(i + 1) / strips);
        }

        GDALClose(threadDS);

        if (fetchId != m_fetchCounter) return; // Abort processing if stale

        if (fetchFailed) {
            emit errorReported("Map Error: No internet connection and cache is empty for this view.");
            emit progressUpdated("Failed", 1.0);
            return;
        }

        emit progressUpdated("Warping Projections...", 0.90);

        // 5. WARP PHASE
        GDALDataset *srcDS = memDriver->Create("", wmPixWidth, wmPixHeight, 3, GDT_Byte, nullptr);
        srcDS->RasterIO(GF_Write, 0, 0, wmPixWidth, wmPixHeight, rawBuffer.data(), wmPixWidth, wmPixHeight, GDT_Byte, 3, bands, 3, wmPixWidth * 3, 1);
        double srcGeo[6] = { wmMinX, wmWidth/wmPixWidth, 0.0, wmMaxY, 0.0, -wmHeight/wmPixHeight };
        srcDS->SetGeoTransform(srcGeo);

        OGRSpatialReference wmSrs;
        wmSrs.importFromEPSG(3857);
        char* wmWkt = nullptr;
        wmSrs.exportToWkt(&wmWkt);
        srcDS->SetProjection(wmWkt);
        CPLFree(wmWkt);

        GDALDataset *dstDS = memDriver->Create("", screenWidth, screenHeight, 3, GDT_Byte, nullptr);
        dstDS->SetProjection(utmWkt.toStdString().c_str());
        double utmMinX = utmCX - (viewW / 2.0);
        double utmMaxY = utmCY + (viewH / 2.0);
        double dstGeo[6] = { utmMinX, viewW/screenWidth, 0.0, utmMaxY, 0.0, -viewH/screenHeight };
        dstDS->SetGeoTransform(dstGeo);

        GDALWarpOptions *warpOpts = GDALCreateWarpOptions();
        warpOpts->hSrcDS = srcDS;
        warpOpts->hDstDS = dstDS;
        warpOpts->eResampleAlg = GRA_Bilinear;
        warpOpts->papszWarpOptions = CSLSetNameValue(warpOpts->papszWarpOptions, "INIT_DEST", "255,255,255");
        warpOpts->pfnTransformer = GDALGenImgProjTransform;
        warpOpts->pTransformerArg = GDALCreateGenImgProjTransformer2(srcDS, dstDS, nullptr);

        GDALWarpOperation warpOp;
        if (warpOp.Initialize(warpOpts) == CE_None) {
            warpOp.ChunkAndWarpImage(0, 0, screenWidth, screenHeight);
        }

        if (warpOpts->pTransformerArg) GDALDestroyGenImgProjTransformer(warpOpts->pTransformerArg);
        GDALDestroyWarpOptions(warpOpts);

        // 6. RENDER PHASE (Sync back to Main UI Thread)
        std::vector<GByte> finalBuffer(screenWidth * screenHeight * 4, 255);
        dstDS->RasterIO(GF_Read, 0, 0, screenWidth, screenHeight, finalBuffer.data(), screenWidth, screenHeight, GDT_Byte, 3, bands, 4, screenWidth * 4, 1);

        GDALClose(srcDS);
        GDALClose(dstDS);

        if (fetchId == m_fetchCounter) {
            // QMetaObject marshals the final image back to the main UI thread safely
            // QMetaObject::invokeMethod(this, [this, finalBuffer, screenWidth, screenHeight, fetchId]() {
            QMetaObject::invokeMethod(this, [this, finalBuffer, screenWidth, screenHeight, fetchId, viewW, viewH, utmCX, utmCY]() {
                if (fetchId == m_fetchCounter) {
                    m_mapBuffer = QImage(finalBuffer.data(), screenWidth, screenHeight, QImage::Format_RGBA8888).copy();

                    // LOCK the render coordinates to this specific image's boundaries
                    m_renderUtmCenterX = utmCX;
                    m_renderUtmCenterY = utmCY;
                    m_renderViewWidth = viewW;
                    m_renderViewHeight = viewH;

                    update();
                    emit progressUpdated("Ready", 1.0);
                }
            }, Qt::QueuedConnection);
        }
    });
}

void MapCanvasItem::commitPan() {
    if (m_visualPanX == 0 && m_visualPanY == 0) return;
    if (boundingRect().width() == 0 || boundingRect().height() == 0) return;

    double mppX = m_viewWidthMeters / boundingRect().width();
    double mppY = m_viewHeightMeters / boundingRect().height();

    double newUtmX = m_utmCenterX - m_visualPanX * mppX;
    double newUtmY = m_utmCenterY + m_visualPanY * mppY;

    m_visualPanX = 0;
    m_visualPanY = 0;

    OGRSpatialReference latLonSrs, utmSrs;
    latLonSrs.SetWellKnownGeogCS("WGS84");
    latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
    utmSrs.importFromWkt(m_utmWkt.toStdString().c_str());
    utmSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);

    OGRCoordinateTransformation *invTrans = OGRCreateCoordinateTransformation(&utmSrs, &latLonSrs);
    if (invTrans) {
        double newLon = newUtmX;
        double newLat = newUtmY;
        invTrans->Transform(1, &newLon, &newLat);
        OCTDestroyCoordinateTransformation(invTrans);

        updateMission(newLat, newLon, m_viewWidthMeters / 1000.0, m_viewHeightMeters / 1000.0);
    } else {
        // [THE FIX]: If GDAL fails to transform, abort the pan and just refresh the map in place
        refreshMapBuffer();
        emit viewChanged();
    }
}

QString MapCanvasItem::estimateDownloadSize() {
    if (m_viewWidthMeters <= 0 || m_viewHeightMeters <= 0 || m_utmWkt.isEmpty()) return "0 KB";

    int screenW = boundingRect().width();
    if (screenW <= 0) return "0 KB";

    // 1. Calculate the exact padded Web Mercator geometry
    double latRad = m_centerLat * M_PI / 180.0;
    double scaleFactor = 1.0 / std::cos(latRad);
    double pad = 3.0; // Matches the 3x padding applied in the downloader

    double wmWidth = m_viewWidthMeters * scaleFactor * pad;
    double wmHeight = m_viewHeightMeters * scaleFactor * pad;

    // 2. Adjust base resolution to account for the padded width
    double currentRes = wmWidth / (screenW * pad);
    const double MIN_RES = 1.0;

    double totalPixels = 0;
    for (double r = currentRes; r >= MIN_RES; r /= 2.0) {
        totalPixels += (wmWidth / r) * (wmHeight / r);
    }

    double estimatedBytes = totalPixels * 0.4;
    double estimatedMB = estimatedBytes / (1024.0 * 1024.0);

    if (estimatedMB < 1.0) {
        return QString::number(estimatedMB * 1024.0, 'f', 1) + " KB";
    }
    return QString::number(estimatedMB, 'f', 1) + " MB";
}

void MapCanvasItem::downloadMapOffline() {
    if (!m_dataset || m_viewWidthMeters <= 0 || m_viewHeightMeters <= 0) return;

    int screenW = boundingRect().width();
    int screenH = boundingRect().height();
    if (screenW <= 0 || screenH <= 0) return;

    QFile xmlFile(":/qt/qml/ui/mapcanvas/osm_config.xml");
    if (!xmlFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
        emit errorReported("Map Error: Failed to load XML configuration.");
        return;
    }
    // QString threadXmlConfig = QTextStream(&xmlFile).readAll().arg(QDir::currentPath() + "/map_cache");
    QString rawXml = QTextStream(&xmlFile).readAll();

    // Dynamically retrieve the cache path from settings
    QSettings settings("Skytec", "MissionPlanner");
    QString defaultPath = QStandardPaths::writableLocation(QStandardPaths::AppDataLocation) + "/map_cache";
    QString cachePath = settings.value("cachePath", defaultPath).toString();
    QDir().mkpath(cachePath); // Guarantee folder exists before GDAL runs

    // Generate the final XML string
    QString threadXmlConfig = rawXml.arg(cachePath);

    double viewW = m_viewWidthMeters;
    double viewH = m_viewHeightMeters;
    double utmCX = m_utmCenterX;
    double utmCY = m_utmCenterY;
    QString utmWkt = m_utmWkt;

    QtConcurrent::run([this, viewW, viewH, utmCX, utmCY, utmWkt, threadXmlConfig, screenW, screenH]() {
        QByteArray xmlBytes = threadXmlConfig.toUtf8();
        GDALDataset* threadDS = (GDALDataset*) GDALOpenEx(xmlBytes.constData(), GDAL_OF_RASTER, nullptr, nullptr, nullptr);
        if (!threadDS) {
            emit errorReported("Map Error: Failed to initialize background downloader.");
            return;
        }

        OGRSpatialReference latLonSrs, utmSrs;
        latLonSrs.SetWellKnownGeogCS("WGS84");
        latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
        utmSrs.importFromWkt(utmWkt.toStdString().c_str());
        utmSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);

        OGRCoordinateTransformation *invTrans = OGRCreateCoordinateTransformation(&utmSrs, &latLonSrs);
        double currentLon = utmCX;
        double currentLat = utmCY;
        if (invTrans) {
            invTrans->Transform(1, &currentLon, &currentLat);
            OCTDestroyCoordinateTransformation(invTrans);
        }

        double webMercCenterX = currentLon * 20037508.34 / 180.0;
        double latRad = currentLat * M_PI / 180.0;
        double webMercCenterY = log(tan(M_PI / 4.0 + latRad / 2.0)) * 20037508.34 / M_PI;

        // 1. CALCULATE THE OFFLINE PANNING BUFFER
        double scaleFactor = 1.0 / std::cos(latRad);
        double pad = 3.0; // Downloads 1 full screen in every direction surrounding your current view

        double wmWidth = viewW * scaleFactor * pad;
        double wmHeight = viewH * scaleFactor * pad;

        double wmMinX = webMercCenterX - (wmWidth / 2.0);
        double wmMaxY = webMercCenterY + (wmHeight / 2.0);

        double wmsGeo[6];
        threadDS->GetGeoTransform(wmsGeo);

        int xOff = static_cast<int>((wmMinX - wmsGeo[0]) / wmsGeo[1]);
        int yOff = static_cast<int>((wmMaxY - wmsGeo[3]) / wmsGeo[5]);
        int dsXSize = static_cast<int>(wmWidth / wmsGeo[1]);
        int dsYSize = static_cast<int>(std::abs(wmHeight / wmsGeo[5]));

        const int CHUNK = 512;
        std::vector<GByte> dummyBuffer(CHUNK * CHUNK * 3);
        int bands[3] = {1, 2, 3};

        // 2. CALIBRATE RESOLUTION TO THE PADDED GEOMETRY
        double currentRes = wmWidth / (screenW * pad);
        const double MIN_RES = 1.0;

        int totalChunks = 0;
        for (double r = currentRes; r >= MIN_RES; r /= 2.0) {
            int targetXPixels = wmWidth / r;
            int targetYPixels = wmHeight / r;
            totalChunks += std::ceil(static_cast<double>(targetXPixels) / CHUNK) *
                           std::ceil(static_cast<double>(targetYPixels) / CHUNK);
        }
        if (totalChunks == 0) totalChunks = 1;

        int currentChunk = 0;
        int layer = 1;

        // 3. EXECUTE PYRAMID FETCH
        for (double r = currentRes; r >= MIN_RES; r /= 2.0) {
            int targetXPixels = wmWidth / r;
            int targetYPixels = wmHeight / r;

            for (int y = 0; y < targetYPixels; y += CHUNK) {
                for (int x = 0; x < targetXPixels; x += CHUNK) {
                    int readXSize = std::min(CHUNK, targetXPixels - x);
                    int readYSize = std::min(CHUNK, targetYPixels - y);

                    int srcXOff = xOff + (x * dsXSize / targetXPixels);
                    int srcYOff = yOff + (y * dsYSize / targetYPixels);
                    int srcXSize = readXSize * dsXSize / targetXPixels;
                    int srcYSize = readYSize * dsYSize / targetYPixels;

                    threadDS->RasterIO(GF_Read, srcXOff, srcYOff, srcXSize, srcYSize,
                                        dummyBuffer.data(), readXSize, readYSize,
                                        GDT_Byte, 3, bands, 3, readXSize * 3, 1);

                    QThread::msleep(50); // Paced to avoid IP ban

                    currentChunk++;
                    double pct = static_cast<double>(currentChunk) / totalChunks;
                    emit progressUpdated(QString("Caching Map Depth %1...").arg(layer), pct);
                }
            }
            layer++;
        }

        GDALClose(threadDS);
        emit progressUpdated("Offline Cache Complete", 1.0);
        emit downloadComplete();
    });
}

void MapCanvasItem::paint(QPainter *painter) {
    if (!m_mapBuffer.isNull()) {
        // Fill the background with a neutral color so edges aren't jarring during a pan
        painter->fillRect(boundingRect(), QColor("#e8edf0"));

        // Draw the image offset by the current mouse drag distance
        painter->drawImage(m_visualPanX, m_visualPanY, m_mapBuffer);
    }

    // 2. Draw the HARP Envelope if data exists
    if (!m_rawEnvelope.isEmpty() && m_viewWidthMeters > 0) {
        int screenWidth = boundingRect().width();
        int screenHeight = boundingRect().height();

        // double mppX = m_viewWidthMeters / screenWidth;
        // double mppY = m_viewHeightMeters / screenHeight;
        // double viewMinX = m_utmCenterX - (m_viewWidthMeters / 2.0);
        // double viewMaxY = m_utmCenterY + (m_viewHeightMeters / 2.0);
        // Use the locked RENDER coordinates, not the live camera coordinates
        double mppX = m_renderViewWidth / screenWidth;
        double mppY = m_renderViewHeight / screenHeight;
        double viewMinX = m_renderUtmCenterX - (m_renderViewWidth / 2.0);
        double viewMaxY = m_renderUtmCenterY + (m_renderViewHeight / 2.0);

        // Setup projection transformers (WGS84 -> Current UTM Zone)
        OGRSpatialReference latLonSrs, utmSrs;
        latLonSrs.SetWellKnownGeogCS("WGS84");
        latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
        utmSrs.importFromWkt(m_utmWkt.toStdString().c_str());
        utmSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);

        OGRCoordinateTransformation *coordTrans = OGRCreateCoordinateTransformation(&latLonSrs, &utmSrs);

        QPolygonF envelopePolygon;

        for (const QVariant& vPt : m_rawEnvelope) {
            QVariantMap pt = vPt.toMap();
            double lon = pt["lon"].toDouble();
            double lat = pt["lat"].toDouble();

            if (coordTrans) {
                coordTrans->Transform(1, &lon, &lat); // Overwrites with UTM X/Y
            }

            // Convert UTM spatial coordinates to Screen Pixels (incorporating active visual pan)
            double screenX = ((lon - viewMinX) / mppX) + m_visualPanX;
            double screenY = ((viewMaxY - lat) / mppY) + m_visualPanY;

            envelopePolygon.append(QPointF(screenX, screenY));
        }

        if (coordTrans) OCTDestroyCoordinateTransformation(coordTrans);

        // Render the polygon with transparency
        painter->setRenderHint(QPainter::Antialiasing);
        painter->setPen(QPen(QColor(255, 0, 0, 200), 2)); // Red border
        painter->setBrush(QColor(255, 0, 0, 50));         // Semi-transparent red fill
        painter->drawPolygon(envelopePolygon);
    }

    // 3. Draw Markers (LZ and DZ)
    if ((m_showLz || m_showDz) && m_renderViewWidth > 0) {
        int screenWidth = boundingRect().width();
        int screenHeight = boundingRect().height();
        double mppX = m_renderViewWidth / screenWidth;
        double mppY = m_renderViewHeight / screenHeight;
        double viewMinX = m_renderUtmCenterX - (m_renderViewWidth / 2.0);
        double viewMaxY = m_renderUtmCenterY + (m_renderViewHeight / 2.0);

        OGRSpatialReference latLonSrs, utmSrs;
        latLonSrs.SetWellKnownGeogCS("WGS84");
        latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
        utmSrs.importFromWkt(m_utmWkt.toStdString().c_str());
        utmSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
        OGRCoordinateTransformation *coordTrans = OGRCreateCoordinateTransformation(&latLonSrs, &utmSrs);

        painter->setRenderHint(QPainter::Antialiasing);

        if (m_showDz && coordTrans) {
            double lon = m_dzLon; double lat = m_dzLat;
            coordTrans->Transform(1, &lon, &lat); // Transforms Lat/Lon to UTM
            double screenX = ((lon - viewMinX) / mppX) + m_visualPanX;
            double screenY = ((viewMaxY - lat) / mppY) + m_visualPanY;

            // Draw infinite flight line
            double rad = m_acHdg * M_PI / 180.0;
            double dx = std::sin(rad);
            double dy = -std::cos(rad);
            QPointF p1(screenX - dx * 3000, screenY - dy * 3000);
            QPointF p2(screenX + dx * 3000, screenY + dy * 3000);

            painter->setPen(QPen(QColor(0, 0, 0, 150), 2, Qt::DashLine));
            painter->drawLine(p1, p2);

            // Draw T-Point rotated to heading
            painter->save();
            painter->translate(screenX, screenY);
            painter->rotate(m_acHdg); // Rotates the painter coordinate system to Aircraft Heading
            painter->setPen(QPen(QColor(255, 140, 0), 3)); // Orange T
            painter->drawLine(-10, -10, 10, -10); // Crossbar (leads flight path)
            painter->drawLine(0, -10, 0, 10);     // Stem

            painter->setPen(QColor(0, 0, 0));
            painter->drawText(15, -15, "DZ");
            painter->restore();
        }

        if (m_showLz && coordTrans) {
            double lon = m_lzLon; double lat = m_lzLat;
            coordTrans->Transform(1, &lon, &lat);
            double screenX = ((lon - viewMinX) / mppX) + m_visualPanX;
            double screenY = ((viewMaxY - lat) / mppY) + m_visualPanY;

            painter->setPen(QPen(QColor(0, 100, 255), 2));
            painter->setBrush(QColor(0, 100, 255, 100));
            painter->drawEllipse(QPointF(screenX, screenY), 7, 7);

            painter->setPen(QColor(0, 0, 0));
            painter->drawText(screenX + 10, screenY + 5, "LZ");
        }

        if (coordTrans) OCTDestroyCoordinateTransformation(coordTrans);
    }

    //

QPointF MapCanvasItem::screenToLatLon(int x, int y) {
    if (m_viewWidthMeters <= 0 || m_viewHeightMeters <= 0 || m_utmWkt.isEmpty()) return QPointF(0, 0);
    int sw = boundingRect().width();
    int sh = boundingRect().height();
    if (sw <= 0 || sh <= 0) return QPointF(0, 0);

    double mppX = m_viewWidthMeters / sw;
    double mppY = m_viewHeightMeters / sh;

    double mapX = x - m_visualPanX;
    double mapY = y - m_visualPanY;

    double utmX = (m_utmCenterX - m_viewWidthMeters / 2.0) + (mapX * mppX);
    double utmY = (m_utmCenterY + m_viewHeightMeters / 2.0) - (mapY * mppY);

    OGRSpatialReference latLonSrs, utmSrs;
    latLonSrs.SetWellKnownGeogCS("WGS84");
    latLonSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);
    utmSrs.importFromWkt(m_utmWkt.toStdString().c_str());
    utmSrs.SetAxisMappingStrategy(OAMS_TRADITIONAL_GIS_ORDER);

    OGRCoordinateTransformation *invTrans = OGRCreateCoordinateTransformation(&utmSrs, &latLonSrs);
    if (invTrans) {
        invTrans->Transform(1, &utmX, &utmY);
        OCTDestroyCoordinateTransformation(invTrans);
        return QPointF(utmX, utmY);
    }

    // [THE FIX]: Safely return 0,0 instead of passing unprojected millions of meters to the UI
    return QPointF(0, 0);
}

QPointF MapCanvasItem::getCenterLatLon() {
    int sw = boundingRect().width();
    int sh = boundingRect().height();
    // Re-use the screen converter to get the exact true center of the canvas
    return screenToLatLon(sw / 2, sh / 2);
}

import QtQuick

import ui

MapCanvasForm {
    id: root

    property alias gdalRenderer: gdalRenderer
    property bool mapInitialized: false

    // ADDED: Delays the GDAL fetch so the GUI window has time to appear on your screen
    Timer {
        id: startupTimer
        interval: 200
        running: false
        repeat: false
        onTriggered: {
            gdalRenderer.updateMission(28.6139, 77.2090, 50, 50)
        }
    }

    MapCanvasItem {
        id: gdalRenderer
        anchors.fill: parent
        z: -1

        // ADD THIS: High-performance QML Crosshair Overlay
        // Rectangle {
        //     width: 20; height: 2; color: "#e53935"
        //     anchors.centerIn: parent
        //     border.color: "white"; border.width: 1
        // }
        // Rectangle {
        //     width: 2; height: 20; color: "#e53935"
        //     anchors.centerIn: parent
        //     border.color: "white"; border.width: 1
        // }
        Item {
            id: crosshair

            property int crossLength: 20
            property int barThickness: 2
            // property color barColor: "#e53935"
            property color barColor: "blue"
            property color strokeColor: "blue"
            property int strokeWidth: 1

            width: crossLength + (strokeWidth * 2)
            height: crossLength + (strokeWidth * 2)
            anchors.centerIn: parent

            Canvas {
                anchors.fill: parent
                antialiasing: true

                onPaint: {
                    var ctx = getContext("2d");
                    ctx.reset();

                    var halfT = crosshair.barThickness / 2;
                    var halfL = crosshair.crossLength / 2;
                    var cx = width / 2;
                    var cy = height / 2;

                    ctx.beginPath();

                    // Top arm
                    ctx.moveTo(cx - halfT, cy - halfL);
                    ctx.lineTo(cx + halfT, cy - halfL);
                    ctx.lineTo(cx + halfT, cy - halfT);

                    // Right arm
                    ctx.lineTo(cx + halfL, cy - halfT);
                    ctx.lineTo(cx + halfL, cy + halfT);
                    ctx.lineTo(cx + halfT, cy + halfT);

                    // Bottom arm
                    ctx.lineTo(cx + halfT, cy + halfL);
                    ctx.lineTo(cx - halfT, cy + halfL);
                    ctx.lineTo(cx - halfT, cy + halfT);

                    // Left arm
                    ctx.lineTo(cx + halfL, cy - halfT); // Adjusted boundary overlap path
                    ctx.lineTo(cx - halfL, cy + halfT);
                    ctx.lineTo(cx - halfL, cy - halfT);
                    ctx.lineTo(cx - halfT, cy - halfT);

                    ctx.closePath();

                    ctx.fillStyle = crosshair.barColor;
                    ctx.fill();

                    ctx.strokeStyle = crosshair.strokeColor;
                    ctx.lineWidth = crosshair.strokeWidth;
                    ctx.lineJoin = "miter";
                    ctx.stroke();
                }
            }
        }

        MouseArea {
            anchors.fill: parent
            hoverEnabled: true

            property int lastX: 0
            property int lastY: 0

            enabled: selectButton.checked || tPointButton.checked

            onPressed: (mouse) => {
                forceActiveFocus()

                if (tPointButton.checked) {
                    let coord = gdalRenderer.screenToLatLon(mouse.x, mouse.y)
                    window.dzLat = coord.y // Y is Lat
                    window.dzLon = coord.x // X is Lon
                    window.hasDz = true
                    window.isDirty = true // Mark project as unsaved

                    let hdg = parseFloat(window.airHeadingField.text) || 0.0
                    gdalRenderer.setDzMarker(window.dzLat, window.dzLon, hdg)

                    selectButton.checked = true // Auto-revert back to the pan tool
                    return
                }

                lastX = mouse.x
                lastY = mouse.y
            }

            onPositionChanged: (mouse) => {
                // // ADD THIS: Instantly update the footer with real-world cursor Lat/Lon
                // let geo = gdalRenderer.screenToGeo(mouse.x, mouse.y)
                // window.coordinateLabel.text = "Lat: " + geo.x.toFixed(5) + " | Lon: " + geo.y.toFixed(5)

                if (pressed) {
                    let dx = mouse.x - lastX
                    let dy = mouse.y - lastY

                    gdalRenderer.panMapPixels(dx, dy)

                    lastX = mouse.x
                    lastY = mouse.y
                }

                // ADD THIS: Dynamic footer coordinate update
                let coord = gdalRenderer.screenToLatLon(mouse.x, mouse.y)
                // Assuming window is the global root ID for MainForm
                window.coordinateLabel.text = "Lat: " + coord.y.toFixed(5) + " | Lon: " + coord.x.toFixed(5)
            }

            // ADD THIS: Finalize the pan and load the new map data
            onReleased: {
                gdalRenderer.commitPan()
            }

            onWheel: (wheel) => {
                if (wheel.angleDelta.y > 0) {
                    gdalRenderer.zoomMap(0.8)
                } else {
                    gdalRenderer.zoomMap(1.25)
                }
            }
        }
    }

    ButtonGroup {
        buttons: [selectButton, drawButton, tPointButton]
    }

    zoomInButton.onClicked: gdalRenderer.zoomMap(0.8)
    zoomOutButton.onClicked: gdalRenderer.zoomMap(1.25)

    onWidthChanged: {
        if (width > 0 && height > 0) {
            if (!mapInitialized) {
                mapInitialized = true
                startupTimer.start() // Trigger the delay
            } else if (!startupTimer.running) {
                gdalRenderer.forceRefresh()
            }
        }
    }
}

    function applyView() {
        let lat = parseFloat(centerLatField.text)
        let lon = parseFloat(centerLonField.text)
        let widthKm = parseFloat(viewWidthField.text)

        if (isNaN(lat) || isNaN(lon) || isNaN(widthKm)) {
            window.showError("Please enter valid numeric coordinates and width.")
            return
        }

        // [THE FIX]: Forcibly drop active focus so the fields can receive live updates again

        centerLonField.focus = false
        viewWidthField.focus = false

        // C++ updateMission automatically handles the jump and re-centers the UTM projection
        window.gdalRenderer.updateMission(lat, lon, widthKm, widthKm)
    }

    applyViewButton.onClicked: applyView()
    centerLatField.onAccepted: applyView()
    centerLonField.onAccepted: applyView()
    viewWidthField.onAccepted: applyView()

    downloadMap.onClicked: {
        downloadMap.text = "Downloading (Background)..."
        downloadMap.enabled = false

        let missionData = {
            "lzLat": parseFloat(latitudeField.text) || 0.0,
            "lzLon": parseFloat(longitudeField.text) || 0.0,
            "lzElev": parseFloat(elevationField.text) || 0.0,
            "jumpAlt": parseFloat(jumpAltField.text) || 0.0,
            "openAlt": parseFloat(canopyAltField.text) || 0.0,
            "weight": parseFloat(weightField.text) || 0.0,
            "acVel": parseFloat(airSpeedField.text) || 0.0,
            "acHdg": parseFloat(airHeadingField.text) || 0.0,
            "canopyVel": parseFloat(canopyGroundSpeedField.text) || 0.0,
            "glideRatio": parseFloat(glideRatioField.text) || 0.0
        }

        // ADD THIS: Harvest the current state of the Wind Profile table
        let windArray = []
        for (let i = 0; i < window.windModel.count; i++) {
            let row = window.windModel.get(i)
            windArray.push({
                "altitude": parseFloat(row.altitude) || 0.0,
                "speed": parseFloat(row.speed) || 0.0,
                "heading": parseFloat(row.heading) || 0.0
            })
        }
        missionData["windProfile"] = windArray

        console.log("Sending parameters to physics engine...")
        MissionController.runMission(window.buildProjectJson())
        // MissionController.runMission(missionData)
    }

    airHeadingField.onTextChanged: {
        if (window.hasDz) {
            window.mapCanvas.gdalRenderer.updateFlightLineHeading(parseFloat(airHeadingField.text) || 0.0)
        }
    }
}

            horizontalAlignment: TextInput.AlignHCenter
            // Critical: Saves manual edits back to the model so C++ gets the changes on Run
            onTextChanged: model.altitude = text
        }
        TextField {
            text: model.speed
            Layout.fillWidth: true
            horizontalAlignment: TextInput.AlignHCenter
            onTextChanged: model.speed = text
        }
        TextField {
            text: model.heading
            Layout.fillWidth: true
            horizontalAlignment: TextInput.AlignHCenter
            onTextChanged: model.heading = text
        }
        Button {
            text: "🗑"
            Layout.preferredWidth: 28
            Layout.preferredHeight: 28
            onClicked: {
                windModel.remove(index)
            }
        }
    }

    addRowButton.onClicked: {
        windModel.append({
            "altitude": "",
            "speed": "",
            "heading": ""
        })
        windListView.positionViewAtEnd()
    }

    fetchButton.onClicked: {
        let lat = parseFloat(window.latitudeField.text)
        let lon = parseFloat(window.longitudeField.text)
        let dateStr = window.dateField.text
        let timeStr = window.timeField.text

        if (isNaN(lat) || isNaN(lon) || dateStr.length !== 10 || timeStr.length !== 5) {
            window.showError("Please enter valid LZ Lat/Lon and Date (YYYY-MM-DD) / Time (HH:MM).")
            return
        }

        // Trigger the C++ fetch thread
        // Prompt for save first, then fetch
        window.requireSaveThen(function() {
            MissionController.fetchWindProfile(lat, lon, dateStr, timeStr)
        })
    }

    // Catch the returned C++ wind data and populate the UI table
    Connections {
        target: MissionController

        function onWindProfileFetched(windData) {
            windModel.clear()

            for (let i = 0; i < windData.length; i++) {
                windModel.append(windData[i])
            }

            windListView.positionViewAtBeginning()
        }
    }
}
